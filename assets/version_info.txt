# PyInstaller version resource for QuoteTracker.exe.
#
# Without this the EXE has no Product name / Company name, so Windows shows
# blank fields in File > Properties and in security prompts.  Filling it in
# does not remove the SmartScreen warning (only code signing does) but it
# makes the file identify itself properly.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(2, 2, 0, 0),
    prodvers=(2, 2, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [StringStruct('CompanyName', 'HAWE Manufacturing US Inc'),
         StringStruct('FileDescription', 'HAWE Quote Tracker - supplier price tracker'),
         StringStruct('FileVersion', '2.2.0.0'),
         StringStruct('InternalName', 'QuoteTracker'),
         StringStruct('OriginalFilename', 'QuoteTracker.exe'),
         StringStruct('ProductName', 'HAWE Quote Tracker'),
         StringStruct('ProductVersion', '2.2.0.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
